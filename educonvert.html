Esquema SQL de la Base de Datos

Estructura requerida en Supabase para el módulo de legislación salvadoreña.

-- ========================================================
-- EDUCONVERT STUDIO / SUPABASE SCHEMA (EduConta SV)
-- ========================================================

CREATE TABLE IF NOT EXISTS public.leyes (
    id TEXT PRIMARY KEY,
    nombre TEXT NOT NULL,
    abreviatura TEXT,
    categoria TEXT NOT NULL,
    decreto_numero TEXT,
    fecha_publicacion DATE,
    descripcion TEXT,
    estado TEXT DEFAULT 'Vigente',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.articulos (
    id TEXT PRIMARY KEY,
    ley_id TEXT REFERENCES public.leyes(id) ON DELETE CASCADE,
    titulo_seccion TEXT,
    capitulo_seccion TEXT,
    seccion TEXT,
    numero TEXT NOT NULL,
    texto TEXT NOT NULL,
    parrafos JSONB DEFAULT '[]'::jsonb,
    orden INTEGER NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_leyes_categoria ON public.leyes(categoria);
CREATE INDEX IF NOT EXISTS idx_articulos_ley_id ON public.articulos(ley_id);
CREATE INDEX IF NOT EXISTS idx_articulos_orden ON public.articulos(orden);

ALTER TABLE public.leyes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.articulos ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Permitir lectura publica de leyes" ON public.leyes FOR SELECT USING (true);
CREATE POLICY "Permitir lectura publica de articulos" ON public.articulos FOR SELECT USING (true);
CREATE POLICY "Permitir insercion admin de leyes" ON public.leyes FOR INSERT WITH CHECK (true);
CREATE POLICY "Permitir actualizacion admin de leyes" ON public.leyes FOR UPDATE USING (true);
CREATE POLICY "Permitir borrado admin de leyes" ON public.leyes FOR DELETE USING (true);
CREATE POLICY "Permitir insercion admin de articulos" ON public.articulos FOR INSERT WITH CHECK (true);
CREATE POLICY "Permitir actualizacion admin de articulos" ON public.articulos FOR UPDATE USING (true);
CREATE POLICY "Permitir borrado admin de articulos" ON public.articulos FOR DELETE USING (true);
